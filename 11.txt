import numpy as np
import random

# Create Grid World
q = np.zeros((16, 4))
start = 0
goal = 15
obstacles = [5, 6, 9, 10]

# Actions: Up, Down, Left, Right
actions = [0, 1, 2, 3]
arrows = ['↑', '↓', '←', '→']

# Environment
def step(state, action):
    row, col = divmod(state, 4)

    if action == 0: row -= 1
    elif action == 1: row += 1
    elif action == 2: col -= 1
    else: col += 1

    if row < 0 or row >= 4 or col < 0 or col >= 4:
        new_state = state
    else:
        new_state = row * 4 + col

    if new_state in obstacles:
        new_state = state

    if new_state == goal:
        return new_state, 10, True

    return new_state, -1, False

# Q-Learning
alpha = 0.1
gamma = 0.9
epsilon = 1.0

for episode in range(5000):
    state = start

    for _ in range(100):

        # Exploration and exploitation
        if random.random() < epsilon:
            action = random.choice(actions)
        else:
            action = np.argmax(q[state])

        next_state, reward, done = step(state, action)

        # Update Q-table
        target = reward
        if not done:
            target += gamma * np.max(q[next_state])

        q[state, action] += alpha * (target - q[state, action])

        state = next_state

        if done:
            break

    epsilon = max(0.05, epsilon * 0.995)

# Evaluate learned policy
success = 0

for episode in range(100):
    state = start

    for _ in range(50):
        action = np.argmax(q[state])
        state, reward, done = step(state, action)

        if done:
            success += 1
            break

print("Success Rate:", success, "%")

# Display optimal policy
for state in range(16):
    if state == goal:
        print("G", end=" ")
    elif state in obstacles:
        print("#", end=" ")
    elif state == start:
        print("S" + arrows[np.argmax(q[state])], end=" ")
    else:
        print(arrows[np.argmax(q[state])], end=" ")

    if state % 4 == 3:
        print()
